% IRLotus ships its bold under the family name "IRlotus" (lowercase L),
% which fontspec's automatic "<family> Bold" probe does not find, so the
% bold face is named explicitly.
%
% Ligatures=PersianTeXNoNumbers swaps xepersian's default `persian-tex-text`
% mapping for the variant that leaves digits alone. Without it every ASCII
% digit typed in Persian text is silently rendered as a Persian digit, while
% digits inside \lr (that is, every measurement in this paper) stay Western
% -- so the same table would print its counts in one numeral system and its
% rates in another. Turning the mapping off makes Western digits the single
% default and keeps the numbers identical to the English edition.
\settextfont[Scale=1.0,BoldFont={IRlotus},

% it is scaled down to sit level with the body text.
\newfontfamily\headfont[Scale=0.86]{IRYekan}
\newfontfamily\displayfont[Scale=0.86]{IRYekan}

% ---------------------------------------------------------------------
% Numbering.
%
% Every digit in this document is Western, structural counters included.
% Two reasons: the measurements are Western throughout -- as they are in
% the English edition, which this paper must stay comparable with -- and
% a document that mixed Persian counters with Western data would make the
% same table read in two numeral systems at once.
%
% The \lr wrapper is what matters for correctness: a compound label such
% as "3.1" is a digit run around a neutral full stop, which the bidi
% algorithm otherwise reorders to "1.3" -- in the heading and in every
% \ref to it. Wrapping \the... rather than patching \ref fixes the
% headings, the cross-references, and the .aux entries in one place.
%
% \theenumi is deliberately left alone: LaTeX expands it in contexts that
% demand a bare number, and a single-digit list label has nothing to
% reorder anyway.
% ---------------------------------------------------------------------
\renewcommand{\thesection}{\lr{\arabic{section}}}
\renewcommand{\thesubsection}{\lr{\arabic{section}.\arabic{subsection}}}
\renewcommand{\thesubsubsection}{%
  \lr{\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}}
\renewcommand{\thefigure}{\lr{\arabic{figure}}}
\renewcommand{\thetable}{\lr{\arabic{table}}}
\renewcommand{\theequation}{\lr{\arabic{equation}}}

% =====================================================================
% TABLE STYLE
%
% Identical grammar to the English edition, so the two documents can be
% read side by side:
%
%   * a tinted header band with the header text in the same ink as the
%     rules, so the reading order of a wide table is unambiguous;
%   * very light zebra banding, which is what makes an eight-column row
%     traceable across the measure;
%   * a stronger tint for group-header and total rows, so the structure
%     of a multi-block table is visible before it is read;
%   * proportional bars beside the headline rates;
%   * semantic colour on the extreme values only -- \hi for the worst
%     outcome in a block, \lo for the best -- never as decoration.
%
% Colours stay legible in greyscale: the tints differ in lightness, and
% every emphasised value is also bold.
% =====================================================================
\definecolor{tabhdr}{HTML}{DCE5EE}   % header band
\definecolor{tabgrp}{HTML}{EDF1F6}   % group-header and total rows
\definecolor{tabrow}{HTML}{F6F8FA}   % zebra band
\definecolor{tabink}{HTML}{15334F}   % header text, rules, headings
\definecolor{barfill}{HTML}{4A7CA8}  % filled portion of a rate bar
\definecolor{bartrack}{HTML}{E2E8EE} % unfilled portion of a rate bar
\definecolor{hiviz}{HTML}{9B2226}    % worst outcome in a block
\definecolor{loviz}{HTML}{1B5E3A}    % best outcome in a block

% Booktabs rules and coloured rows fight each other: the default
% \aboverulesep / \belowrulesep leave an uncoloured gap at every rule.
% Zeroing them and paying the space back through \extrarowheight keeps
% the bands flush against the rules.

\setlength{\belowrulesep}{0pt}
\setlength{\extrarowheight}{1.7pt}
\setlength{\heavyrulewidth}{1.0pt}
\setlength{\lightrulewidth}{0.45pt}
\setlength{\cmidrulewidth}{0.4pt}
\arrayrulecolor{tabink}

\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.14}

\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}

% ---------------------------------------------------------------------
% Bidirectional helpers.
%
% Every technical term in this paper stays in English and every measured
% quantity stays in Western digits. Both are LTR runs inside RTL text, so
% they are wrapped rather than left to the bidi algorithm: \num in
% particular keeps compound values such as 12/12 or 5.6x from being
% reordered around their separators.
% ---------------------------------------------------------------------
\newcommand{\en}[1]{\lr{#1}}                       % English term
\newcommand{\num}[1]{\lr{#1}}                      % numeric quantity
% "a becomes b". Both operands and the arrow live in one LTR run, so the
% arrow keeps pointing from the reported value to the ablated one; left to
% the bidi algorithm the arrow is a neutral between two digit runs and can
% end up separating them in the wrong order.
\newcommand{\numto}[2]{\lr{#1 $\rightarrow$ #2}}
% \code drops a step, as it does in the English edition: Consolas is set
% at the size of the surrounding table body rather than above it, which is
% what keeps a long identifier inside a narrow cell.
\newcommand{\code}[1]{\lr{\texttt{\footnotesize #1}}}  % identifier / path
\newcommand{\bk}{\discretionary{}{}{}}             % zero-width break point

% Column widths as fractions of \textwidth, so they track the text block.
% The subtracted \tabcolsep pays for this column's share of the
% inter-column padding, which guarantees the row cannot exceed the measure.
\newcommand{\tw}[1]{\dimexpr #1\textwidth-2\tabcolsep\relax}

% A proportional bar for a rate in [0,1], drawn as a filled segment over a
% full-width track. Forced LTR: the bar is a quantity, not text, and must
% grow in the same direction on every page.
\newlength{\barwd}
\setlength{\barwd}{12mm}
\newcommand{\asrbar}[1]{%
  \lr{\makebox[\barwd][l]{%
    \textcolor{bartrack}{\rule[-0.15ex]{\barwd}{5pt}}%
    \hspace*{-\barwd}%
    \textcolor{barfill}{\rule[-0.15ex]{\dimexpr#1\barwd\relax}{5pt}}}}%
}

% ---------------------------------------------------------------------
% Figures are generated artefacts. Until they have been built, draw a
% labelled placeholder instead of failing the build. Scaled to fit both
% the text width and a maximum height, so a diagram with an extreme
% aspect ratio cannot run off the page.
% ---------------------------------------------------------------------
\newcommand{\safefig}[2][1.0]{%
  \IfFileExists{#2}%
    {\includegraphics[width=#1\textwidth,height=0.82\textheight,keepaspectratio]{#2}}%
    {\fbox{\parbox[c][2.6cm][c]{0.92\textwidth}{\centering
       \lr{\ttfamily\footnotesize #2}\\[4pt]
       ساخته نشده --- سازندهٔ شکل‌ها را اجرا کنید}}}%
}

% --- Abstract --------------------------------------------------------
% Tinted panel rather than IEEEtran's run-in abstract: at this measure a

\begin{center}
{\setlength{\fboxsep}{9pt}%
\colorbox{tabrow}{%
\begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}
\vspace{3pt}
{\headfont\bfseries\color{tabink}چکیده}\par\vspace{2pt}
\small
ارزیابی‌های امنیتی مدل‌های زبانی بزرگ معمولاً خودِ مدل را می‌کاوند.
شکست‌هایی که در استقرار واقعی اهمیت دارند اما معمولاً به برنامهٔ پیرامون
مدل تعلق دارند --- \en{input filter}، \en{retriever}،
\en{output rewriter}، رمزگشای فایل، مجری ابزار --- زیرا همین خط لوله است
که به یک کانال دادهٔ غیرقابل‌اعتماد، اقتدارِ یک کانال دستور را می‌بخشد.
این مقاله ارزیابی‌ای را گزارش می‌کند که برنامه را هدف می‌گیرد. ما نُه سطح
\en{HTTP} از یک فروشگاه عمداً آسیب‌پذیرِ مبتنی بر \en{LLM} را به شکل
\en{Garak generator} بسته‌بندی می‌کنیم، دوازده \en{detector} می‌افزاییم که
به‌جای یک امضای سبک‌شناختی، \emph{نقض مشخصی از سیاست را در برابر
\en{ground truth}} می‌آزمایند، و فراداده‌های خودِ برنامه به همراه رازِ هر
تلاش را از مسیر کانال \en{notes} در \en{Garak} حمل می‌کنیم تا امتیازدهی
دقیق باشد و امتناع از قضاوت جای اطمینان کاذب را بگیرد. هر مؤلفه یک
\en{Garak plugin} درجه‌یک است؛ هستهٔ \en{Garak} دست‌نخورده می‌ماند و هر
اندازه‌گیری از طریق \en{CLI} استاندارد قابل بازپخش است. پنج \en{suite}
شامل \num{28} اجرا و \num{678} \en{attempt} ارزیابی‌شده،
\en{prompt injection} (مستقیم و غیرمستقیم)، \en{information disclosure}
و \en{data poisoning} را پوشش می‌دهد. \en{data poisoning}
پرمخاطره‌ترین دسته است (\num{$\mathrm{ASR}=0.348$}). سخت‌کردن یک
\en{system prompt} در پنج \en{persona}، نشت کوپن را به‌صورت یکنواخت از
\num{0.472} به \num{0.083} می‌رساند بی‌آنکه هرگز آن را حذف کند، و ثابت
نگه‌داشتنِ پیکرهٔ حمله در ده پیکربندی \en{guardrail} نشان می‌دهد دو مورد از
ده مورد، بهتر از نبودِ هیچ \en{guardrail} عمل نمی‌کنند. موفقیت در
تکنیک‌هایی متمرکز است که هرگز هدف خود را نام نمی‌برند:
\en{fictional framing} در \num{14/15} تلاش موفق شد در برابر \num{1/15}
برای پرسیدنِ ساده، و در برابر سخت‌گیرترین \en{persona}، پرسش‌هایی که
\emph{قواعد} را می‌خواستند \num{5.6} برابر بیشتر از پرسش‌هایی که خودِ راز را
می‌خواستند به رازِ محافظت‌شده رسیدند. پنج نظر با برچسب نادرست یک
\en{backdoor} هدفمند در طبقه‌بند می‌سازند که رانش احتمال آن، چهار گامِ
بودجه پیش از تغییر هر برچسبی قابل اندازه‌گیری است. \en{retrieval} با
برچسب اعتماد، بازیابی غیرقابل‌اعتماد را کاملاً حذف کرد
(\numto{12/12}{0/12})، و بازرسی نشان می‌دهد نرخ باقی‌ماندهٔ
سطحِ پاسخ، مثبت کاذبِ \en{detector} است. یک \en{detector} عمومی و
استاندارد که در کنار ابزار ما اجرا شد، در برابر سیاست واقعی برنامه به
\en{recall} برابر \num{1.000} با \en{precision} برابر \num{0.293} رسید. یک
مطالعهٔ \en{ablation}، سهم هر لایهٔ دفاعی را از سهم هر جزء دستگاه
اندازه‌گیری تفکیک می‌کند و نشان می‌دهد که حذف کانال \en{ground truth}
اعداد این مطالعه را تضعیف نمی‌کند، بلکه آن‌ها را از میان برمی‌دارد.
